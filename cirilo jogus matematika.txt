# --------------------------------------------------
# QUESTION GENERATOR
# --------------------------------------------------
def generate_question():
    question_type = random.choice([1, 2, 3, 4])

    # x + a = b
    if question_type == 1:
        x = random.randint(2, 30)
        a = random.randint(2, 20)
        b = x + a

        return {
            "question": f"x + {a} = {b}",
            "answer": x
        }

    # x - a = b
    elif question_type == 2:
        x = random.randint(10, 40)
        a = random.randint(2, x - 1)
        b = x - a

        return {
            "question": f"x - {a} = {b}",
            "answer": x
        }

    # a × x = b
    elif question_type == 3:
        x = random.randint(2, 12)
        a = random.randint(2, 10)
        b = a * x

        return {
            "question": f"{a} × x = {b}",
            "answer": x
        }

    # x ÷ a = b
    else:
        a = random.randint(2, 10)
        b = random.randint(2, 12)
        x = a * b

        return {
            "question": f"x ÷ {a} = {b}",
            "answer": x
        }


# --------------------------------------------------
# INITIALIZE GAME
# --------------------------------------------------
if "started" not in st.session_state:
    st.session_state.started = False

if "question_number" not in st.session_state:
    st.session_state.question_number = 1

if "score" not in st.session_state:
    st.session_state.score = 0

if "questions" not in st.session_state:
    st.session_state.questions = []

if "answered" not in st.session_state:
    st.session_state.answered = False

if "feedback" not in st.session_state:
    st.session_state.feedback = ""

# --------------------------------------------------
# TITLE
# --------------------------------------------------
st.markdown(
    '<div class="main-title">🧮 Equation Quest</div>',
    unsafe_allow_html=True
)

st.markdown(
    '<div class="subtitle">🎯 Grade 5 Mathematics Challenge</div>',
    unsafe_allow_html=True
)

# --------------------------------------------------
# START SCREEN
# --------------------------------------------------
if not st.session_state.started:

    st.info(
        "Welcome to Equation Quest! "
        "Solve 10 equations and collect as many points as possible."
    )

    st.markdown("### 🎮 How to Play")

    st.write("🔹 Solve the equation by finding the value of **x**.")
    st.write("🔹 Each correct answer = **2 points**.")
    st.write("🔹 There are **10 questions**.")
    st.write("🔹 Maximum score = **20 points**.")
    st.write("🔹 Try to earn the highest badge! 🏆")

    if st.button("🚀 START GAME", use_container_width=True):

        st.session_state.started = True
        st.session_state.question_number = 1
        st.session_state.score = 0
        st.session_state.questions = [
            generate_question() for _ in range(10)
        ]
        st.session_state.answered = False
        st.session_state.feedback = ""

        st.rerun()

# --------------------------------------------------
# GAME SCREEN
# --------------------------------------------------
else:

    total_questions = 10
    current = st.session_state.question_number

    # SCORE
    st.markdown(
        f"""
        <div class="score-box">
        ⭐ Score: {st.session_state.score} / 20
        &nbsp;&nbsp;&nbsp; 
        📝 Question: {current} / {total_questions}
        </div>
        """,
        unsafe_allow_html=True
    )

    # PROGRESS
    progress = (current - 1) / total_questions
    st.progress(progress)

    # CURRENT QUESTION
    question_data = st.session_state.questions[current - 1]

    st.markdown(
        f"""
        <div class="question-box">
        ❓ Solve for x<br><br>
        {question_data["question"]}
        </div>
        """,
        unsafe_allow_html=True
    )

    # --------------------------------------------------
    # ANSWER FORM
    # --------------------------------------------------
    if not st.session_state.answered:

        answer = st.number_input(
            "✏️ What is the value of x?",
            min_value=0,
            max_value=1000,
            step=1,
            key=f"answer_{current}"
        )

        if st.button(
            "✅ CHECK ANSWER",
            use_container_width=True
        ):

            correct_answer = question_data["answer"]

            if answer == correct_answer:

                st.session_state.score += 2

                st.session_state.feedback = (
                    f"🎉 Excellent! {answer} is correct! "
                    f"You earned **2 points**."
                )

                st.session_state.answered = True

            else:

                st.session_state.feedback = (
                    f"❌ Not quite. Try to solve the equation carefully. "
                    f"Remember: find the number that makes both sides equal."
                )

                st.session_state.answered = True

            st.rerun()

    # --------------------------------------------------
    # FEEDBACK
    # --------------------------------------------------
    else:

        if "🎉" in st.session_state.feedback:
            st.success(st.session_state.feedback)
        else:
            st.warning(st.session_state.feedback)

            st.info(
                f"💡 Correct answer: "
                f"**x = {question_data['answer']}**"
            )

        # NEXT QUESTION
        if current < total_questions:

            if st.button(
                "➡️ NEXT QUESTION",
                use_container_width=True
            ):

                st.session_state.question_number += 1
                st.session_state.answered = False
                st.session_state.feedback = ""

                st.rerun()

        else:

            # --------------------------------------------------
            # FINAL RESULT
            # --------------------------------------------------
            st.balloons()

            final_score = st.session_state.score

            st.markdown("## 🏆 GAME COMPLETE!")

            st.markdown(
                f"""
                <div class="success-message">
                🎯 Your Final Score<br>
                ⭐ {final_score} / 20
                </div>
                """,
                unsafe_allow_html=True
            )

            # BADGES
            if final_score == 20:

                st.success(
                    "🏆 GOLD EQUATION MASTER!\n\n"
                    "Amazing! You solved every equation correctly!"
                )

            elif final_score >= 16:

                st.success(
                    "🥇 EQUATION CHAMPION!\n\n"
                    "Excellent mathematical thinking!"
                )

            elif final_score >= 12:

                st.info(
                    "🥈 EQUATION EXPLORER!\n\n"
                    "Great work! Keep practicing!"
                )

            elif final_score >= 8:

                st.warning(
                    "🥉 EQUATION LEARNER!\n\n"
                    "Good effort! You are improving!"
                )

            else:

                st.warning(
                    "🌱 EQUATION BEGINNER!\n\n"
                    "Keep practicing and you will become stronger!"
                )

            # PLAY AGAIN
            if st.button(
                "🔄 PLAY AGAIN",
                use_container_width=True
            ):

                st.session_state.started = False
                st.session_state.question_number = 1
                st.session_state.score = 0
                st.session_state.questions = []
                st.session_state.answered = False
                st.session_state.feedback = ""

                st.rerun()
